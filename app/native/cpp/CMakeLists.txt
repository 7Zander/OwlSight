cmake_minimum_required(VERSION 3.24)
project(OwlSightNative VERSION 0.2.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(OPENEXR_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(OPENEXR_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(OPENEXR_BUILD_PYTHON OFF CACHE BOOL "" FORCE)
set(OPENEXR_INSTALL OFF CACHE BOOL "" FORCE)
set(OPENEXR_INSTALL_PKG_CONFIG OFF CACHE BOOL "" FORCE)
set(OPENEXR_FORCE_INTERNAL_IMATH ON CACHE BOOL "" FORCE)
set(OPENEXR_FORCE_INTERNAL_DEFLATE ON CACHE BOOL "" FORCE)
set(OPENEXR_FORCE_INTERNAL_OPENJPH ON CACHE BOOL "" FORCE)
set(IMATH_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(IMATH_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(IMATH_INSTALL OFF CACHE BOOL "" FORCE)
set(IMATH_INSTALL_PKG_CONFIG OFF CACHE BOOL "" FORCE)
set(JSON_BuildTests OFF CACHE BOOL "" FORCE)
set(JSON_Install OFF CACHE BOOL "" FORCE)
include(FetchContent)
FetchContent_Declare(Imath
  URL https://github.com/AcademySoftwareFoundation/Imath/releases/download/v3.2.2/Imath-3.2.2.tar.gz
  URL_HASH SHA256=0f5a783b424f374e6f27ec8b0c73130e89b08814ac8fa2e84fd7fe0b05862c53)
FetchContent_MakeAvailable(Imath)
FetchContent_Declare(OpenEXR
  URL https://github.com/AcademySoftwareFoundation/openexr/releases/download/v3.4.15/openexr-3.4.15.tar.gz
  URL_HASH SHA256=ab893d8003773ccd9a5556b2caf38da591ae37e20b06ee9d589a08984c5191f2)
FetchContent_MakeAvailable(OpenEXR)
# Header and license are vendored from nlohmann/json v3.12.0.
add_executable(owlsight-exr main.cpp exr.cpp preview.cpp)
target_include_directories(owlsight-exr PRIVATE ../../third_party/nlohmann)
target_link_libraries(owlsight-exr PRIVATE OpenEXR::OpenEXR)
if(WIN32)
  target_compile_definitions(owlsight-exr PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(owlsight-exr PRIVATE psapi)
endif()
if(MSVC)
  target_compile_options(owlsight-exr PRIVATE /utf-8 /W4 /EHsc)
else()
  target_compile_options(owlsight-exr PRIVATE -Wall -Wextra -Wpedantic)
  if(MINGW)
    target_link_options(owlsight-exr PRIVATE -static)
  endif()
endif()
install(TARGETS owlsight-exr RUNTIME DESTINATION .)

# The registry helper is launched only by the explicit application settings action.
if(WIN32)
  add_executable(owlsight-platform platform.cpp)
  target_compile_definitions(owlsight-platform PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(owlsight-platform PRIVATE advapi32 shell32)
  if(MSVC)
    target_compile_options(owlsight-platform PRIVATE /utf-8 /W4 /EHsc)
  elseif(MINGW)
    target_compile_options(owlsight-platform PRIVATE -Wall -Wextra -Wpedantic)
    target_link_options(owlsight-platform PRIVATE -static -municode)
  endif()
  install(TARGETS owlsight-platform RUNTIME DESTINATION .)
endif()
option(OWLSIGHT_BUILD_DESKTOP "Build the standalone Win32 / ImGui / D3D12 client" OFF)
if(OWLSIGHT_BUILD_DESKTOP)
  enable_language(RC)
  # Resolve the static zlib target before OCIO/minizip transitive discovery.
  # -static alone does not override an explicit DLL import archive.
  set(ZLIB_USE_STATIC_LIBS ON)
  find_package(ZLIB REQUIRED)
  if(MINGW)
    foreach(zlib_path IN ITEMS ZLIB_LIBRARY ZLIB_LIBRARY_RELEASE ZLIB_LIBRARY_DEBUG)
      if("${${zlib_path}}" MATCHES "[.]dll[.]a$")
        message(FATAL_ERROR "OwlSight requires static zlib; use scripts/build-native.ps1 to set all zlib library paths.")
      endif()
    endforeach()
  endif()
  find_package(OpenColorIO 2.5.1 EXACT CONFIG REQUIRED)
  add_executable(OwlSight WIN32
    ../desktop/host.cpp ../desktop/app.cpp ../desktop/core.cpp ../desktop/renderer.cpp
    ../desktop/version.rc exr.cpp preview.cpp color/main.cpp platform.cpp
    "${OWLSIGHT_IMGUI_SOURCE}/imgui.cpp" "${OWLSIGHT_IMGUI_SOURCE}/imgui_draw.cpp"
    "${OWLSIGHT_IMGUI_SOURCE}/imgui_tables.cpp" "${OWLSIGHT_IMGUI_SOURCE}/imgui_widgets.cpp"
    "${OWLSIGHT_IMGUI_SOURCE}/backends/imgui_impl_win32.cpp" "${OWLSIGHT_IMGUI_SOURCE}/backends/imgui_impl_dx12.cpp")
  target_include_directories(OwlSight PRIVATE
    "${OWLSIGHT_IMGUI_SOURCE}" "${OWLSIGHT_IMGUI_SOURCE}/backends" ../../third_party/nlohmann)
  target_compile_definitions(OwlSight PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE OWLSIGHT_OCIO_LIBRARY OWLSIGHT_PLATFORM_LIBRARY)
  target_link_libraries(OwlSight PRIVATE OpenEXR::OpenEXR OpenColorIO::OpenColorIO
    d3d12 dxgi d3dcompiler dwmapi imm32 shlwapi comdlg32 ole32 uuid shell32 advapi32 bcrypt psapi)
  if(MSVC)
    target_compile_options(OwlSight PRIVATE /utf-8 /W4 /EHsc)
  elseif(MINGW)
    target_compile_options(OwlSight PRIVATE -Wall -Wextra -Wpedantic)
    target_link_options(OwlSight PRIVATE -static -municode)
  endif()
endif()
