cmake_minimum_required(VERSION 3.24)
project(OwlSightQtQuick VERSION 0.2.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(OWLSIGHT_BUILD_DESKTOP OFF CACHE BOOL "" FORCE)
add_subdirectory(../cpp native-core)
find_package(Qt6 6.11.3 EXACT REQUIRED COMPONENTS Quick QuickControls2 ShaderTools GuiPrivate)
set(ZLIB_USE_STATIC_LIBS ON)
find_package(ZLIB REQUIRED)
find_package(OpenColorIO 2.5.1 EXACT CONFIG REQUIRED)
qt_add_executable(OwlSight WIN32 main.cpp viewer.cpp viewer.hpp core.cpp core.hpp gpu_canvas.cpp gpu_canvas.hpp gpu_frame.hpp upload_payload.cpp upload_payload.hpp
 ../cpp/exr.cpp ../cpp/preview.cpp ../cpp/color/main.cpp)
target_include_directories(OwlSight PRIVATE ../../third_party/nlohmann)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../../release.json" OWLSIGHT_RELEASE_JSON)
string(JSON OWLSIGHT_BUILD_NUMBER GET "${OWLSIGHT_RELEASE_JSON}" build)
target_compile_definitions(OwlSight PRIVATE OWLSIGHT_BUILD_NUMBER=${OWLSIGHT_BUILD_NUMBER} NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE OWLSIGHT_OCIO_LIBRARY)
target_link_libraries(OwlSight PRIVATE Qt6::Quick Qt6::QuickControls2 Qt6::GuiPrivate OpenEXR::OpenEXR OpenColorIO::OpenColorIO psapi shlwapi bcrypt)
qt_add_resources(OwlSight "qml" PREFIX "/" FILES qml/Main.qml)
qt_add_shaders(OwlSight "preview_shaders" PREFIX "/gpu" BASE shaders GLSL "330,300es" HLSL 50 MSL 12
 FILES shaders/preview.vert shaders/preview.frag)
