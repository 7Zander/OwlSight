cmake_minimum_required(VERSION 3.24)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../../release.json" OWLSIGHT_RELEASE_JSON)
string(JSON OWLSIGHT_VERSION GET "${OWLSIGHT_RELEASE_JSON}" version)
project(OwlSightQtQuick VERSION ${OWLSIGHT_VERSION} LANGUAGES C CXX RC)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(OWLSIGHT_BUILD_TOOLS OFF CACHE BOOL "Build legacy command-line helpers" FORCE)
add_subdirectory(../cpp native-core)
find_package(Qt6 6.11.3 EXACT REQUIRED COMPONENTS Quick QuickControls2 ShaderTools GuiPrivate Network)
set(ZLIB_USE_STATIC_LIBS ON)
find_package(ZLIB REQUIRED)
find_package(OpenColorIO 2.5.1 EXACT CONFIG REQUIRED)
option(OWLSIGHT_ENABLE_SMOKE_DRIVER "Include the developer-only smoke driver" OFF)
qt_add_executable(OwlSight WIN32 main.cpp update_manager.cpp update_manager.hpp viewer.cpp viewer.hpp core.cpp core.hpp prefetch_budget.cpp prefetch_budget.hpp gpu_canvas.cpp gpu_canvas.hpp gpu_frame.hpp upload_payload.cpp upload_payload.hpp
 ../cpp/color/main.cpp)
if(OWLSIGHT_ENABLE_SMOKE_DRIVER)
 target_sources(OwlSight PRIVATE diagnostics/smoke_driver.cpp)
 target_compile_definitions(OwlSight PRIVATE OWLSIGHT_ENABLE_SMOKE_DRIVER)
endif()
target_include_directories(OwlSight PRIVATE ../../third_party/nlohmann)
string(JSON OWLSIGHT_BUILD_NUMBER GET "${OWLSIGHT_RELEASE_JSON}" build)
target_compile_definitions(OwlSight PRIVATE OWLSIGHT_BUILD_NUMBER=${OWLSIGHT_BUILD_NUMBER} OWLSIGHT_VERSION="${OWLSIGHT_VERSION}" NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE OWLSIGHT_OCIO_LIBRARY)
target_link_libraries(OwlSight PRIVATE Qt6::Quick Qt6::QuickControls2 Qt6::GuiPrivate Qt6::Network owlsight-image OpenColorIO::OpenColorIO shlwapi bcrypt shell32)
if(MINGW)
 target_link_options(OwlSight PRIVATE $<$<CONFIG:Release>:-Wl,--strip-all>)
endif()
qt_add_resources(OwlSight "qml" PREFIX "/" FILES qml/Main.qml)
qt_add_shaders(OwlSight "preview_shaders" PREFIX "/gpu" BASE shaders GLSL "330,300es" HLSL 50 MSL 12
 FILES shaders/preview.vert shaders/preview.frag)

set(OWLSIGHT_ICON_ICO "${CMAKE_CURRENT_SOURCE_DIR}/../../resources/owlsight.ico")
set(OWLSIGHT_RESOURCE_NAME "OwlSight")
set(OWLSIGHT_RESOURCE_DESCRIPTION "OwlSight EXR viewer")
configure_file(windows-resource.rc.in "${CMAKE_CURRENT_BINARY_DIR}/OwlSight.rc" @ONLY)
target_sources(OwlSight PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/OwlSight.rc")
add_executable(OwlSightUpdater WIN32 updater/launcher.c)
target_compile_definitions(OwlSightUpdater PRIVATE UNICODE _UNICODE)
target_link_libraries(OwlSightUpdater PRIVATE bcrypt shell32)
if(MINGW)
 target_link_options(OwlSightUpdater PRIVATE -municode $<$<CONFIG:Release>:-Wl,--strip-all>)
endif()
set(OWLSIGHT_RESOURCE_NAME "OwlSightUpdater")
set(OWLSIGHT_RESOURCE_DESCRIPTION "OwlSight update launcher")
configure_file(windows-resource.rc.in "${CMAKE_CURRENT_BINARY_DIR}/OwlSightUpdater.rc" @ONLY)
target_sources(OwlSightUpdater PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/OwlSightUpdater.rc")
add_dependencies(OwlSight OwlSightUpdater)
qt_add_resources(OwlSight "branding" PREFIX "/branding" BASE "../../resources" FILES "../../resources/owlsight.svg")
